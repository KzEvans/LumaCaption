cmake_minimum_required(VERSION 3.24)
project(lumawhisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "CPU baseline; enable only in separately verified build")
set(GGML_BACKEND_DL OFF CACHE BOOL "Package backends statically in lumawhisper" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(NOT WHISPER_SOURCE)
  include(FetchContent)
  FetchContent_Declare(whisper GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG v1.8.1 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(whisper)
else()
  add_subdirectory("${WHISPER_SOURCE}" whisper)
endif()
add_library(lumawhisper SHARED luma_whisper.cpp)
target_link_libraries(lumawhisper PRIVATE whisper)
set_target_properties(lumawhisper PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
install(TARGETS lumawhisper LIBRARY DESTINATION . RUNTIME DESTINATION .)
